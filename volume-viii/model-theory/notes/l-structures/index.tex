\section{Languages and Structures}

\begin{toolkitbox}{L-Structures Toolkit --- Quick Reference}
\small
\begin{tabular}{l l l}
\toprule
\textbf{Concept} & \textbf{Symbol / Notation} & \textbf{Detail} \\
\midrule
First-order language    & $\mathcal{L}$                          & \hyperref[def:first-order-language]{$\downarrow$} \\
L-structure             & $\mathcal{M} = (M, \mathcal{L}^{\mathcal{M}})$ & \hyperref[def:l-structure]{$\downarrow$} \\
Domain / universe       & $|\mathcal{M}|$ or $M$                 & \hyperref[def:l-structure]{$\downarrow$} \\
Interpretation of $f$   & $f^{\mathcal{M}} : M^n \to M$          & \hyperref[def:l-structure]{$\downarrow$} \\
Interpretation of $R$   & $R^{\mathcal{M}} \subseteq M^n$        & \hyperref[def:l-structure]{$\downarrow$} \\
Interpretation of $c$   & $c^{\mathcal{M}} \in M$                & \hyperref[def:l-structure]{$\downarrow$} \\
Variable assignment     & $s : \mathrm{Var} \to M$               & \hyperref[def:variable-assignment]{$\downarrow$} \\
Term evaluation         & $s(t) \in M$                           & \hyperref[def:term-evaluation]{$\downarrow$} \\
Satisfaction            & $\mathcal{M} \models \varphi[s]$       & \hyperref[def:satisfaction]{$\downarrow$} \\
\bottomrule
\end{tabular}
\end{toolkitbox}
